package com.nkds.hosikoouma.j

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.appwidget.AppWidgetManager
import android.content.ComponentName
import android.content.Context
import android.content.Intent
import android.graphics.Bitmap
import android.graphics.BitmapFactory
import android.graphics.drawable.BitmapDrawable
import android.os.Handler
import android.os.Looper
import androidx.localbroadcastmanager.content.LocalBroadcastManager
import java.nio.charset.Charset
import androidx.media3.common.Metadata
import androidx.annotation.OptIn
import androidx.media3.common.AudioAttributes
import androidx.media3.common.C
import androidx.media3.common.MediaMetadata
import androidx.media3.common.Player
import androidx.media3.common.util.UnstableApi
import androidx.media3.datasource.DefaultDataSource
import androidx.media3.datasource.DefaultHttpDataSource
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.exoplayer.source.DefaultMediaSourceFactory
import androidx.media3.session.MediaLibraryService
import androidx.media3.session.MediaSession
import androidx.media3.extractor.metadata.icy.IcyInfo
import coil.ImageLoader
import coil.request.ImageRequest
import coil.request.SuccessResult

private const val CHANNEL_ID = "jasmine_channel"

@OptIn(UnstableApi::class)
class PlaybackService : MediaLibraryService() {

    private var mediaLibrarySession: MediaLibrarySession? = null
    private lateinit var player: ExoPlayer
    private lateinit var settingsManager: SettingsManager
    
    private val handler = Handler(Looper.getMainLooper())
    private val CROSSFADE_DURATION = 2500L
    private val FADE_UPDATE_INTERVAL = 16L
    
    private var fadeRunnable: Runnable? = null
    private var isFadingOut = false
    
    // Кэш текущей обложки и ID трека для предотвращения мерцания
    private var currentArtworkBitmap: Bitmap? = null
    private var currentMediaId: String? = null

    private val libraryCallback = object : MediaLibrarySession.Callback {}

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            "ACTION_PLAY_PAUSE" -> { if (player.isPlaying) player.pause() else player.play() }
            "ACTION_NEXT" -> { player.seekToNext() }
            "ACTION_PREV" -> { player.seekToPrevious() }
        }
        // Убрали прямой вызов updateWidget(), так как слушатели плеера сделают это чище
        return super.onStartCommand(intent, flags, startId)
    }

    override fun onCreate() {
        super.onCreate()
        settingsManager = SettingsManager(this)
        createNotificationChannel()
        
        val audioAttributes = AudioAttributes.Builder()
            .setUsage(C.USAGE_MEDIA)
            .setContentType(C.AUDIO_CONTENT_TYPE_MUSIC)
            .build()

        val httpDataSourceFactory = DefaultHttpDataSource.Factory()
            .setUserAgent("JasminePlayer/1.0")
            .setDefaultRequestProperties(mapOf("Icy-MetaData" to "1"))
            .setAllowCrossProtocolRedirects(true)

        val smartDataSourceFactory = MediaCacheManager.createSmartDataSourceFactory(this, httpDataSourceFactory)

        player = ExoPlayer.Builder(this)
            .setAudioAttributes(audioAttributes, true)
            .setHandleAudioBecomingNoisy(true)
            .setMediaSourceFactory(DefaultMediaSourceFactory(this).setDataSourceFactory(smartDataSourceFactory))
            .build()

        player.addListener(object : Player.Listener {
            override fun onMediaItemTransition(mediaItem: androidx.media3.common.MediaItem?, reason: Int) {
                if (settingsManager.isCrossfadeEnabled) {
                    animateVolume(from = player.volume, to = 1.0f, duration = CROSSFADE_DURATION)
                } else {
                    cancelFade()
                    player.volume = 1.0f
                }
                isFadingOut = false
                updateWidget()
            }

            override fun onIsPlayingChanged(isPlaying: Boolean) {
                // Если изменилось только состояние игры - обновляем ТОЛЬКО иконку
                updateWidget(onlyPlayStatus = true)
            }

            override fun onMetadata(metadata: Metadata) {
                try {
                    val candidate = extractStreamTitleFromMetadata(metadata)
                    if (!candidate.isNullOrBlank()) {
                        val fixed = fixEncodingIfNeeded(candidate)
                        updateCurrentMediaItemMetadata(player, fixed)
                        val intent = Intent("com.nkds.jasmine.RADIO_METADATA_UPDATE")
                        intent.putExtra("title", fixed)
                        LocalBroadcastManager.getInstance(this@PlaybackService).sendBroadcast(intent)
                        updateWidget()
                    }
                } catch (e: Exception) {}
            }
        })

        startPositionMonitoring()

        val sessionActivityPendingIntent = packageManager?.getLaunchIntentForPackage(packageName)?.let {
            PendingIntent.getActivity(
                this,
                0,
                it.apply { putExtra("open_player_screen", true) },
                PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
            )
        }

        mediaLibrarySession = MediaLibrarySession.Builder(this, player, libraryCallback)
            .setSessionActivity(sessionActivityPendingIntent!!)
            .build()
    }

    private fun updateWidget(onlyPlayStatus: Boolean = false) {
        val currentItem = player.currentMediaItem
        val title = currentItem?.mediaMetadata?.title?.toString() ?: "Not playing"
        val artist = currentItem?.mediaMetadata?.artist?.toString() ?: "Jasmine Player"
        val isPlaying = player.isPlaying
        val mediaId = currentItem?.mediaId
        
        val appWidgetManager = AppWidgetManager.getInstance(this)
        val componentName = ComponentName(this, PlayerWidget::class.java)
        val ids = appWidgetManager.getAppWidgetIds(componentName)
        if (ids.isEmpty()) return

        // ОПТИМИЗАЦИЯ: Если изменился только статус Play/Pause и трек тот же
        if (onlyPlayStatus && mediaId == currentMediaId) {
            ids.forEach { id ->
                PlayerWidget.updatePlayStatus(this, appWidgetManager, id, isPlaying)
            }
            return
        }

        // Если это новый трек - сбрасываем кэш обложки
        if (mediaId != currentMediaId) {
            currentArtworkBitmap = null
            currentMediaId = mediaId
        }

        val metadata = currentItem?.mediaMetadata
        val artworkData = metadata?.artworkData
        val artworkUri = metadata?.artworkUri

        // Пытаемся получить Bitmap мгновенно (из байтов)
        if (currentArtworkBitmap == null && artworkData != null) {
            try {
                currentArtworkBitmap = BitmapFactory.decodeByteArray(artworkData, 0, artworkData.size)
            } catch (e: Exception) {}
        }

        if (currentArtworkBitmap != null) {
            ids.forEach { id ->
                PlayerWidget.updateWidget(this, appWidgetManager, id, title, artist, isPlaying, currentArtworkBitmap)
            }
        } else if (artworkUri != null) {
            // Обновляем текст, не мерцая кнопкой (используем частичное обновление)
            ids.forEach { id ->
                PlayerWidget.updateMetadata(this, appWidgetManager, id, title, artist, isPlaying)
            }
            
            val imageLoader = ImageLoader(this)
            val request = ImageRequest.Builder(this)
                .data(artworkUri)
                .target(
                    onSuccess = { result ->
                        val bitmap = (result as? BitmapDrawable)?.bitmap
                        if (bitmap != null) {
                            currentArtworkBitmap = bitmap
                            ids.forEach { id ->
                                PlayerWidget.updateWidget(this, appWidgetManager, id, title, artist, isPlaying, bitmap)
                            }
                        }
                    },
                    onError = {
                        ids.forEach { id ->
                            PlayerWidget.updateWidget(this, appWidgetManager, id, title, artist, isPlaying, null)
                        }
                    }
                )
                .build()
            imageLoader.enqueue(request)
        } else {
            ids.forEach { id ->
                PlayerWidget.updateWidget(this, appWidgetManager, id, title, artist, isPlaying, null)
            }
        }
    }

    private fun animateVolume(from: Float, to: Float, duration: Long) {
        cancelFade()
        val startTime = System.currentTimeMillis()
        fadeRunnable = object : Runnable {
            override fun run() {
                val elapsed = System.currentTimeMillis() - startTime
                val rawProgress = (elapsed.toFloat() / duration).coerceIn(0f, 1f)
                val progress = if (rawProgress < 0.5f) 2 * rawProgress * rawProgress else 1 - (-2 * rawProgress + 2) * (-2 * rawProgress + 2) / 2
                val currentVolume = from + (to - from) * progress
                player.volume = currentVolume
                if (rawProgress < 1f) handler.postDelayed(this, FADE_UPDATE_INTERVAL) else { player.volume = to; fadeRunnable = null }
            }
        }
        handler.post(fadeRunnable!!)
    }

    private fun cancelFade() {
        fadeRunnable?.let { handler.removeCallbacks(it) }
        fadeRunnable = null
    }

    private fun startPositionMonitoring() {
        handler.post(object : Runnable {
            override fun run() {
                if (settingsManager.isCrossfadeEnabled && player.isPlaying && player.duration > 0 && !isFadingOut) {
                    val remainingTime = player.duration - player.currentPosition
                    if (remainingTime <= CROSSFADE_DURATION && !player.isCurrentMediaItemLive) {
                        isFadingOut = true
                        animateVolume(from = player.volume, to = 0f, duration = CROSSFADE_DURATION)
                    }
                }
                handler.postDelayed(this, 500)
            }
        })
    }

    private fun updateCurrentMediaItemMetadata(player: Player, streamTitle: String) {
        val currentItem = player.currentMediaItem ?: return
        val metadataBuilder = currentItem.mediaMetadata.buildUpon()
        if (streamTitle.contains(" - ")) {
            val parts = streamTitle.split(" - ", limit = 2)
            metadataBuilder.setArtist(parts[0].trim())
            metadataBuilder.setTitle(parts[1].trim())
        } else {
            metadataBuilder.setTitle(streamTitle)
        }
        val updatedMetadata = metadataBuilder.build()
        if (updatedMetadata != currentItem.mediaMetadata) {
            val updatedItem = currentItem.buildUpon().setMediaMetadata(updatedMetadata).build()
            player.replaceMediaItem(player.currentMediaItemIndex, updatedItem)
        }
    }

    private fun extractStreamTitleFromMetadata(metadata: Metadata): String? {
        val candidates = mutableListOf<String>()
        for (i in 0 until metadata.length()) {
            val entry = metadata[i]
            val cls = entry?.javaClass
            if (cls != null) {
                val methodNames = listOf("getTitle", "getStreamTitle", "getText", "getString", "getValue")
                for (m in methodNames) {
                    try {
                        val method = cls.getMethod(m)
                        val res = method.invoke(entry)?.toString()
                        if (!res.isNullOrBlank()) candidates.add(res.trim())
                    } catch (_: Throwable) {}
                }
                val fieldNames = listOf("title", "streamTitle", "text", "value")
                for (f in fieldNames) {
                    try {
                        val field = cls.getDeclaredField(f)
                        field.isAccessible = true
                        val v = field.get(entry)?.toString()
                        if (!v.isNullOrBlank()) candidates.add(v.trim())
                    } catch (_: Throwable) {}
                }
            }
            try {
                val s = entry?.toString() ?: ""
                if (s.isNotBlank()) {
                    val p1 = Regex("""StreamTitle\s*=\s*'([^']*)'""", RegexOption.IGNORE_CASE)
                    p1.find(s)?.let { candidates.add(it.groupValues[1].trim()) }
                    val p2 = Regex("""title\s*=\s*['"]?([^;'\"]+)['"]?""", RegexOption.IGNORE_CASE)
                    p2.find(s)?.let { candidates.add(it.groupValues[1].trim()) }
                }
            } catch (_: Throwable) {}
        }

        if (candidates.isEmpty()) return null
        val cyr = candidates.find { containsCyrillic(it) }
        if (!cyr.isNullOrBlank()) return cleanCandidate(cyr)
        val dash = candidates.find { it.contains(" - ") }
        if (!dash.isNullOrBlank()) return cleanCandidate(dash)
        return cleanCandidate(candidates.first())
    }

    private fun cleanCandidate(s: String): String {
        var res = s.trim()
        if ((res.startsWith("'") && res.endsWith("'")) || (res.startsWith("\"") && res.endsWith("\""))) {
            res = res.substring(1, res.length - 1)
        }
        try {
            res = res.replace(Regex("\\b(?:url|href|title)\\s*=\\s*[^;,'\"\\s]+", RegexOption.IGNORE_CASE), "").trim()
            val badKeys = listOf("url", "href", "sid", "br", "icy-name", "icy-description", "icy-url", "content", "id", "name", "song", "streamtitle")
            val keysPattern = Regex(badKeys.joinToString("|") { Regex.escape(it) } + "\\s*=\\s*[^;,'\\\"\\s]+", RegexOption.IGNORE_CASE)
            res = res.replace(keysPattern, "").trim()
            res = res.replace(Regex("[;,:]+"), " - ").replace(Regex("\\s+"), " ").trim()
        } catch (_: Throwable) {}
        return res
    }

    private fun containsCyrillic(s: String): Boolean {
        for (ch in s) if (ch in '\u0400'..'\u04FF') return true
        return false
    }

    private fun fixEncodingIfNeeded(s: String): String {
        if (containsCyrillic(s)) return s
        return try {
            val decoded = String(s.toByteArray(Charsets.ISO_8859_1), Charset.forName("CP1251"))
            if (containsCyrillic(decoded)) decoded else s
        } catch (e: Throwable) { s }
    }

    override fun onGetSession(controllerInfo: MediaSession.ControllerInfo): MediaLibrarySession? = mediaLibrarySession

    override fun onDestroy() {
        cancelFade()
        handler.removeCallbacksAndMessages(null)
        mediaLibrarySession?.run {
            player.release()
            release()
            mediaLibrarySession = null
        }
        super.onDestroy()
    }

    private fun createNotificationChannel() {
        val channel = NotificationChannel(CHANNEL_ID, "Jasmine Player", NotificationManager.IMPORTANCE_LOW)
        val notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        notificationManager.createNotificationChannel(channel)
    }
}
